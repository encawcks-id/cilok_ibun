Kode Google Apps Script (code.gs)

Salin kode di bawah dan tempelkan ke Google Apps Script Spreadsheet Anda.

Cara Setup Google Apps Script:
  1. Buka Google Spreadsheet baru di Google Drive Anda.
  2. Klik menu Extensions (Ekstensi) > Apps Script.
  3. Hapus semua kode bawaan, lalu salin (copy) seluruh kode di bawah ini dan paste ke file Code.gs.
  4. Klik Deploy > New deployment. Pilih type Web app.
  5. Set Execute as: Me dan Who has access: Anyone (Siapa saja).
  6. Klik Deploy, izinkan otorisasi (Authorize Access), lalu salin Web App URL yang dihasilkan ke kolom URL Pengaturan di aplikasi ini.
function doGet(e) {
  var action = e.parameter.action;
  if (action === "read") {
    return ContentService.createTextOutput(JSON.stringify(readAllData()))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "GAS Web App Active" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    if (data.action === "sync") {
      saveAllData(data.payload);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Data tersinkronisasi ke Google Sheets" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function saveAllData(payload) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Save Settings
  if(payload.settings) {
    var sheetSettings = getOrCreateSheet(ss, "Settings");
    sheetSettings.clear();
    sheetSettings.appendRow(["Key", "Value"]);
    for (var key in payload.settings) {
      sheetSettings.appendRow([key, payload.settings[key]]);
    }
  }

  // Save Products
  if(payload.products) {
    var sheetProd = getOrCreateSheet(ss, "Products");
    sheetProd.clear();
    sheetProd.appendRow(["ID", "Name", "Category", "Unit", "Price", "IsActive"]);
    payload.products.forEach(function(p) {
      sheetProd.appendRow([p.id, p.name, p.category, p.unit, p.price, p.isActive]);
    });
  }

  // Save Customers
  if(payload.customers) {
    var sheetCust = getOrCreateSheet(ss, "Customers");
    sheetCust.clear();
    sheetCust.appendRow(["ID", "Name", "Phone", "Address"]);
    payload.customers.forEach(function(c) {
      sheetCust.appendRow([c.id, c.name, c.phone, c.address || ""]);
    });
  }

  // Save Orders
  if(payload.orders) {
    var sheetOrd = getOrCreateSheet(ss, "Orders");
    sheetOrd.clear();
    sheetOrd.appendRow(["ID", "PO Number", "Date", "Shipping Date", "Customer Name", "Phone", "Total", "Payment Status", "Payment Method", "Amount Paid", "Status", "Items JSON"]);
    payload.orders.forEach(function(o) {
      sheetOrd.appendRow([
        o.id, o.poNumber, o.date, o.shippingDate, o.customerName, o.customerPhone,
        o.total, o.paymentStatus, o.paymentMethod, o.amountPaid, o.status, JSON.stringify(o.items || [])
      ]);
    });
  }
}

function readAllData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var result = { settings: {}, products: [], customers: [], orders: [] };

  // Read Settings
  var sheetSettings = ss.getSheetByName("Settings");
  if(sheetSettings) {
    var vals = sheetSettings.getDataRange().getValues();
    for(var i = 1; i < vals.length; i++) {
      if(vals[i][0]) result.settings[vals[i][0]] = vals[i][1];
    }
  }

  // Read Products
  var sheetProd = ss.getSheetByName("Products");
  if(sheetProd) {
    var vals = sheetProd.getDataRange().getValues();
    for(var i = 1; i < vals.length; i++) {
      result.products.push({ id: vals[i][0], name: vals[i][1], category: vals[i][2], unit: vals[i][3], price: Number(vals[i][4]), isActive: Boolean(vals[i][5]) });
    }
  }

  // Read Customers
  var sheetCust = ss.getSheetByName("Customers");
  if(sheetCust) {
    var vals = sheetCust.getDataRange().getValues();
    for(var i = 1; i < vals.length; i++) {
      result.customers.push({ id: vals[i][0], name: vals[i][1], phone: vals[i][2], address: vals[i][3] });
    }
  }

  // Read Orders
  var sheetOrd = ss.getSheetByName("Orders");
  if(sheetOrd) {
    var vals = sheetOrd.getDataRange().getValues();
    for(var i = 1; i < vals.length; i++) {
      var items = [];
      try { items = JSON.parse(vals[i][11]); } catch(e){}
      result.orders.push({
        id: vals[i][0], poNumber: vals[i][1], date: vals[i][2], shippingDate: vals[i][3],
        customerName: vals[i][4], customerPhone: vals[i][5], total: Number(vals[i][6]),
        paymentStatus: vals[i][7], paymentMethod: vals[i][8], amountPaid: Number(vals[i][9]),
        status: vals[i][10], items: items
      });
    }
  }

  return result;
}

function getOrCreateSheet(ss, name) {
  var s = ss.getSheetByName(name);
  if (!s) s = ss.insertSheet(name);
  return s;
}
